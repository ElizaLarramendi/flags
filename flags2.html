// Map each file code to its human-readable country/region name and colors
flagOptions = flagCodes
  .map(code => {
    const colors = flagColors[code] || ["White"];
    const secondary = colors.slice(1).sort();
    return {
      label: code.startsWith("gb-")
        ? ({ "gb-eng": "England", "gb-nir": "Northern Ireland", "gb-sct": "Scotland", "gb-wls": "Wales" })[code]
        : regionNames.of(code.toUpperCase()),
      value: code,
      primary: colors[0],
      secondary: secondary,
      sharedKey: secondary.length ? secondary.join(" + ") : "Solid (no secondary color)",
      allColors: colors
    };
  })
  .sort((a, b) => a.label.localeCompare(b.label))

Flag Mosaic (Made of Tiny Flags)

viewof mosaicCols = Inputs.range([20, 60], {
  label: "Mosaic grid columns:",
  step: 2,
  value: 38
})

// Sample the selected flag's pixels and replace each cell with the closest-colored mini flag
mosaicGrid = {
  const img = new Image();
  img.src = `w2560/${selectedFlag.value}.png`;
  await img.decode();

  const cols = mosaicCols;
  const aspect = img.naturalHeight / img.naturalWidth;
  const rows = Math.max(10, Math.round(cols * aspect));

  const canvas = document.createElement("canvas");
  canvas.width = cols;
  canvas.height = rows;
  const ctx = canvas.getContext("2d");
  ctx.fillStyle = "#ffffff";
  ctx.fillRect(0, 0, cols, rows);
  ctx.drawImage(img, 0, 0, cols, rows);

  const data = ctx.getImageData(0, 0, cols, rows).data;
  const entries = Object.entries(flagRGB);
  const cells = [];

  for (let i = 0; i < data.length; i += 4) {
    const r = data[i], g = data[i + 1], b = data[i + 2];
    let bestCode = "us";
    let bestDist = Infinity;
    for (const [code, [fr, fg, fb]] of entries) {
      const dist = (r - fr) ** 2 + (g - fg) ** 2 + (b - fb) ** 2;
      if (dist < bestDist) {
        bestDist = dist;
        bestCode = code;
      }
    }
    const matchOption = flagOptions.find(d => d.value === bestCode);
    cells.push({code: bestCode, label: matchOption ? matchOption.label : bestCode});
  }

  return html`<div style="max-width: 680px">
    <div style="
      display: grid;
      grid-template-columns: repeat(${cols}, 1fr);
      gap: 1px;
      background: #cbd5e1;
      border: 1px solid #94a3b8;
      border-radius: 4px;
      overflow: hidden;
    ">
      ${cells.map(c => html`<img
        src="w2560/${c.code}.png"
        title="${c.label}"
        style="width: 100%; aspect-ratio: 3 / 2; object-fit: cover; display: block"
      >`)}
    </div>
    <p style="font-size: 0.85rem; color: #555; margin-top: 0.4rem">
      Mosaic of <strong>${selectedFlag.label}</strong> built from ${cols * rows} miniature flags (${cols} × ${rows}). Hover over any tile to see which flag is used!
    </p>
  </div>`;
}
